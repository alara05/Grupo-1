CREATE USER JOTEL IDENTIFIED BY HOSPEDAJE;
GRANT CONNECT, RESOURCE, UNLIMITED TABLESPACE TO HOSPEDAJE123;
CONNECT HOSPEDAJE/HOSPEDAJE123;

CREATE TABLE CIUDADES(
ID_CIU VARCHAR2(10) PRIMARY KEY,
NOM_CIU VARCHAR2(20) NOT NULL UNIQUE,
PRO_PER VARCHAR2(20) NOT NULL
);

CREATE TABLE CLIENTES(
CED_CLI VARCHAR2(10) PRIMARY KEY,
NOM_CLI VARCHAR2(20) NOT NULL,
APE_CLI VARCHAR2(20) NOT NULL,
ID_CIU_PER VARCHAR2(10) NOT NULL REFERENCES CIUDADES(ID_CIU),
CEL_CLI VARCHAR2(10) NOT NULL UNIQUE,
CHECK (LENGTH(CED_CLI)=10),
CHECK (LENGTH(CEL_CLI)=10)
);

CREATE TABLE EMPLEADOS(
CED_EMP VARCHAR2(10) PRIMARY KEY,
NOM_EMP VARCHAR2(20) NOT NULL,
APE_EMP VARCHAR2(20) NOT NULL,
CEL_EMP VARCHAR2(10) NOT NULL UNIQUE,
CHECK (LENGTH(CED_EMP)=10),
CHECK (LENGTH(CEL_EMP)=10)
);

CREATE TABLE HABITACIONES(
ID_HAB VARCHAR2(10) PRIMARY KEY,
NOM_HAB VARCHAR2(20) NOT NULL UNIQUE,
TIP_HAB VARCHAR2(20) NOT NULL,
COSXDIA NUMBER(10,2) NOT NULL,
EST_HAB VARCHAR2(1) NOT NULL,
CHECK (COSXDIA>0),
CHECK (EST_HAB IN ('L','O','M'))
);

CREATE TABLE FACTURA_HOSPEDAJE(
NUM_HOS NUMBER PRIMARY KEY,
FEC_ING DATE NOT NULL,
ID_CLI_ING VARCHAR2(10) NOT NULL REFERENCES CLIENTES(CED_CLI),
ID_EMP_ING VARCHAR2(10) NOT NULL REFERENCES EMPLEADOS(CED_EMP),
ID_HAB_HOS VARCHAR2(10) NOT NULL REFERENCES HABITACIONES(ID_HAB),
EST_FAC VARCHAR2(10) NOT NULL,
TOTAL NUMBER(10,2) NOT NULL,
FEC_SAL DATE,
ID_EMP_SAL VARCHAR2(10) REFERENCES EMPLEADOS(CED_EMP),
CHECK (TOTAL>=0),
CHECK (FEC_SAL IS NULL OR FEC_SAL>=FEC_ING),
CHECK (EST_FAC IN ('ACTIVA','PAGADA','ANULADA'))
);

CREATE TABLE SERVICIOS(
ID_SER VARCHAR2(10) PRIMARY KEY,
NOM_SER VARCHAR2(20) NOT NULL UNIQUE,
COSXUNI NUMBER(10,2) NOT NULL,
CHECK (COSXUNI>0)
);

CREATE TABLE DETALLE_USO_SERVICIOS(
COD_SER_USA VARCHAR2(10) REFERENCES SERVICIOS(ID_SER),
CANTIDAD NUMBER NOT NULL,
FEC_USA DATE NOT NULL,
ID_EMP_ATE VARCHAR2(10) REFERENCES EMPLEADOS(CED_EMP),
NUM_HOS_CAR NUMBER REFERENCES FACTURA_HOSPEDAJE(NUM_HOS),
CHECK (CANTIDAD>0),
UNIQUE (COD_SER_USA,FEC_USA,ID_EMP_ATE,NUM_HOS_CAR)
);
