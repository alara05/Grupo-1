Ejercicios Oracle Interactivos 🚀

Primero intenta → luego mira SQL + INSERT → después valida si está bien o mal

0. Crear Usuario y Permisos

CREATE USER biblioteca IDENTIFIED BY 1234;
GRANT CONNECT, RESOURCE TO biblioteca;

1. PRIMARY KEY

CREATE TABLE usuarios (
 id NUMBER,
 nombre VARCHAR2(100)
);
CREATE TABLE usuarios (
 id NUMBER PRIMARY KEY,
 nombre VARCHAR2(100) NOT NULL
);

INSERT INTO usuarios VALUES (1, 'Juan');
INSERT INTO usuarios VALUES (1, 'Pedro');

❌ Incorrecto: La PRIMARY KEY no permite valores duplicados.

2. UNIQUE

CREATE TABLE clientes (
 correo VARCHAR2(100)
);
CREATE TABLE clientes (
 id NUMBER PRIMARY KEY,
 correo VARCHAR2(100) UNIQUE
);

INSERT INTO clientes VALUES (1, 'a@gmail.com');
INSERT INTO clientes VALUES (2, 'a@gmail.com');

❌ Incorrecto: UNIQUE no permite valores duplicados.

3. CHECK

CREATE TABLE personas (
 edad NUMBER
);
CREATE TABLE personas (
 id NUMBER PRIMARY KEY,
 edad NUMBER CHECK (edad >= 18)
);

INSERT INTO personas VALUES (1, 15);

❌ Incorrecto: No cumple la condición CHECK (edad >= 18).

4. FOREIGN KEY

CREATE TABLE pedidos (
 cliente_id NUMBER
);
CREATE TABLE clientes (
 id NUMBER PRIMARY KEY
);

CREATE TABLE pedidos (
 id NUMBER PRIMARY KEY,
 cliente_id NUMBER,
 CONSTRAINT fk_cliente FOREIGN KEY (cliente_id)
 REFERENCES clientes(id)
);

INSERT INTO pedidos VALUES (1, 99);

❌ Incorrecto: El cliente no existe (violación de clave foránea).

5. CASCADE

FOREIGN KEY (cliente_id) REFERENCES clientes(id);
FOREIGN KEY (cliente_id) REFERENCES clientes(id) ON DELETE CASCADE;

DELETE FROM clientes WHERE id = 1;

⚠️ Puede fallar si NO usas CASCADE y existen registros relacionados.

6. DEFAULT

estado VARCHAR2(20);
estado VARCHAR2(20) DEFAULT 'ACTIVO';

INSERT INTO productos (id, nombre) VALUES (1, 'Laptop');

✔ Correcto: Se asigna automáticamente 'ACTIVO'.

7. NOT NULL

nombre VARCHAR2(100);
nombre VARCHAR2(100) NOT NULL;

INSERT INTO empleados (id, nombre) VALUES (1, NULL);

❌ Incorrecto: NOT NULL no permite valores nulos.

8. ALTER TABLE

ALTER empleados salario NUMBER;
ALTER TABLE empleados ADD salario NUMBER;

❌ Incorrecto: falta TABLE en la sintaxis.
